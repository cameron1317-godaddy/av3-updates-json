      A tuple of (GREEN_HIGHLIGHT, CLOSER) where:
        GREEN_HIGHLIGHT: String containing the green highlight tag syntax or an empty string
        CLOSER:          String containing the closer tag syntax or an empty string
    """
    if use_highlights:
      return GREEN_HIGHLIGHT, CLOSER
    return '', ''

  def increment_total(self):
    """
    Increment the number of total artifacts cleaned by 1.
    """
    self.total_artifacts += 1

  def increment_total_sigs(self):
    """
    Increment the number of total unique signatures found by 1.
    """
    self.total_signatures += 1

  def update_impacted_locations(self, dir):
    """
    Update impacted locations array.

    Args:
      dir: String containing unique location

    Returns:
      None
    """
    self.impacted_locations.add(dir)

  def update_sig_families(self, family):
    """
    Update total malware signature family set.

    Args:
      family: String containing unique malware family

    Returns:
      None
    """
    self.unique_sig_families.add(family)

  def update_malware_overview(self, subtype): 
    """
    Updates overview message for signatures under the 'malware' branch based on sub-branch to add variety

    Args:
      subtype: String containing the malware subtype

    Returns:
      A string containing updated overview message
    """
    overview = 'Causes unintended behavior'
    if not subtype:
      return overview
    elif 'redirect' in subtype:
      overview = 'Causes unintended redirects'
    elif 'stealer' in subtype or 'cc-' in subtype:
      overview = 'Steals sensitive information'
    elif 'security-bypass' in subtype:
      overview = 'Bypasses security implementations'
    return overview
    
  def normalize_location(self, location, slash_count, value):
    """
    Updates directory name based on outermost directory name

    Args:
      location:    String containing the outermost directory name
      slash_count: Integer containing the number of slashes present in the directory path
      value:       String containing the full directory path

    Returns:
      A string containing the outer directory name
    """
    if location == '.' and slash_count >= 1:
      location = value.split('/')[1]
    elif location == '..' and slash_count >= 1:
      location = '../' + value.split('/')[1]
    elif slash_count == 0:
      # e.g., wp_posts.post_content, ID = 2862. if no comma is present, most likely a file/dir with no starting `./`
      loc_details = location.split(',')
      location = f'<{loc_details[0]}>' if len(loc_details) > 1 else f'./{loc_details[0]}'

    location = re.sub(r'\,\s+ID\s+\=\s+\d+', '', location) # Remove unique user IDs for db-cleanup output

    if location == '.' or location == '..':
      location += '/'

    return location
  
  def split_sig_details(self, sig):
    """
    Separate the malware signature by the dot notation that it's curated with.

    Args:
      sig: String containing malware signature.

    Returns:
      A tuple containing each individual detail that makes up the malware signature
    """

    sig_lang = None
    sig_type = None
    sig_subtype = None
    sig_group = None
    sig_subgroup = None

    try:
      details = sig.split('.')
      details_num = len(details)

      sig_lang = details[0]
      sig_type = details[1]

      if details_num > 4: # e.g., php.malware.obfuscated.211.003
        sig_subtype = details[2]
        sig_group = details[3]
        sig_subgroup = details[4]
      elif details_num == 4:  # e.g., php.malware.obfuscated.211
        sig_subtype = details[2]
        sig_group = details[3]
      elif details_num == 3:  # e.g., image.php_backdoor.136
        sig_group = details[2] # only one group, last index is number value
      
      return sig_lang, sig_type, sig_subtype, sig_group, sig_subgroup
    
    except IndexError as e:
      return sig_lang, sig_type, sig_subtype, sig_group, sig_subgroup

  def get_output(self):
    """
    Parse the cleanup output file.

    Args:
      path: String containing path to cleanup output file.

    Returns:
      An array containing each individual cleanup output entry as a string
    """
    result = []

    try:
      with open(self.source, 'r') as output:

        for line in output:
          line = line.strip()

          if line:
            if CLEARED_TEXT in line or LISTCLEANED_IDENTIFIER in line: # line starts with `CLEARED` keyword

              regex = r'(?:CLEARED:\s+Cleared|File\s+fixed|Database\s+entry\s+fixed).*(?:file|database|review\)|removed\)): '
              line = re.sub(regex, '', line) # remove redundant CLEARED kyewords/phrasing

              line = line.replace('..//', '../') # normalize double-slash bug output from scripts
              line = re.sub(r'\$up\$', '..', line) # convert $up$ to ../
              line = line.replace('Details: Removed row', 'Details: removed_user.backdoor.001') # manual/generic removed db user signature
              line = line.replace('Found malicious DB user: ', '')

              if DETAIL_SEPERATOR in line: # line includes signature details
                result.append(line)
              else: # line does not include signature details
                new_line = re.sub(r'\[\d+-\d+-\d+\s\d+\:\d+\:\d+\]', 'Details: custom.malware.generic.001', line)

                if new_line != line: # uses listcleaned formatting (e.g., [mm-dd-yyyy hh:mm:ss UTC])
                  result.append(new_line)

                elif self.show_all: # some other custom cleared output that can't be accounted for
                  self.other_output_items.append(line)

            elif self.show_all: # all other items that do not start with `CLEARED` keyword
              self.other_output_items.append(line)

      return result
    
    except (FileNotFoundError, PermissionError) as e:
      print(f'\n{ERROR_HIGHLIGHT}Could not open source output file -> {self.source}{CLOSER}\n')
      raise

  def sort_output(self, output):
    """
    Sort the output results based on their malware signature.

    Args:
      output: Array containing strings of cleanup output.

    Returns:
      A dict containing each unique malware signature and the corresponding file paths with those signatures
    """
    sorted_output = {}

    for line in output:

      details = line.split(DETAIL_SEPERATOR)
      
      key = details[1]
      value = details[0]

      if key and value:

        key = re.sub(r'Removed malicious DB user:\s+', '', key)
        key = re.sub(r':\s+\w+$', '', key)

        if sorted_output.get(key):

          sorted_output[key]['hits'].append(value)
          sorted_output[key]['count'] += 1

          location = self.normalize_location(value.split('/')[0], value.count('/'), value)
          
          if location not in sorted_output[key]['locations']:
            sorted_output[key]['locations'].append(location)
          
          self.update_impacted_locations(location)
          self.increment_total()

        else:

          location = self.normalize_location(value.split('/')[0], value.count('/'), value)

          sorted_output[key] = {
            'hits': [value],
            'count': 1,
            'locations': [location]
          }

          self.update_impacted_locations(location)

          malware_family = re.sub(r'(\.\d+){,3}$', '', key)
          self.update_sig_families(malware_family)

          self.increment_total()
          self.increment_total_sigs()

    self.sorted_output = sorted_output.items()

  def normalize_variant(self, raw_variant_value):
    """
    Normalizes the variant value to be more human-readable

    Args:
      raw_variant_value: String containing the signatures raw variation value

    Returns:
      variant: A string containing the human-readable version of the signature's variant value
    """
    if not raw_variant_value:
      return '0 (None)'

    try:
      variant = int(raw_variant_value)
    except ValueError as e:
      return '0 (None)'
    
    if variant == 0:
      return '0 (None)'
    
    last_two_digits = variant % 100
    last_digit = variant % 10
    
    if 11 <= last_two_digits <= 13:
      return f'{variant}th'
    elif last_digit == 1:
      return f'{variant}st'
    elif last_digit == 2:
      return f'{variant}nd'
    elif last_digit == 3:
      return f'{variant}rd'
    else:
      return f'{variant}th'

  def format_output(self, use_highlights):
    """
    Properly formats output for display and storage.

    Args:
      use_highlights: Boolean containing the use of highlights

    Returns:
      A string containing the parsed and formatted output
    """
    formatted_output = ''
    HIGHLIGHT_TAG, CLOSING_TAG = self._get_tags(use_highlights)

    for sig, files in self.sorted_output:
      sig_component_num = sig.count('.') + 1
      formatted_output += f"\n{HIGHLIGHT_TAG}Signature:{CLOSING_TAG}   {sig}"

      sig_lang, sig_type, sig_subtype, sig_group, sig_subgroup = self.split_sig_details(sig)

      if self.verbose:
        formatted_output += f"\n{HIGHLIGHT_TAG}Language:{CLOSING_TAG}    {SIG_DETAILS.get('lang').get(sig_lang, 'N/A')}"

      formatted_output += f"\n{HIGHLIGHT_TAG}Category:{CLOSING_TAG}    {SIG_DETAILS.get('category').get(sig_type, 'N/A')}"

      if sig_type == 'malware':
            formatted_output += f"\n{HIGHLIGHT_TAG}Overview:{CLOSING_TAG}    {self.update_malware_overview(sig_subtype)}"
      else:
        formatted_output += f"\n{HIGHLIGHT_TAG}Overview:{CLOSING_TAG}    {SIG_DETAILS.get('overview').get(sig_type, 'N/A')}"

      if self.verbose:
        if sig_component_num >= 4 and not re.match(r'^\d+$', sig_subtype):
          formatted_output += f"\n{HIGHLIGHT_TAG}Description:{CLOSING_TAG} {SIG_DETAILS.get('description').get(sig_subtype, 'N/A')}"

      if self.experimental: # experimental option
        sig_frequency = self.calculate_frequency(sig_group, sig_subgroup)
        formatted_output += f"\n{HIGHLIGHT_TAG}Frequency:{CLOSING_TAG}   {sig_frequency}"

      if self.verbose:
        formatted_output += f"\n{HIGHLIGHT_TAG}Variant:{CLOSING_TAG}     {self.normalize_variant(sig_subgroup) or '0 (None)'}"


      if self.verbose:
        formatted_output += f"\n{HIGHLIGHT_TAG}Status:{CLOSING_TAG}      CLEARED"

      if self.verbose:
        formatted_output += f"\n{HIGHLIGHT_TAG}Locations:{CLOSING_TAG}   {files['locations']}"

      formatted_output += f"\n{HIGHLIGHT_TAG}Cleaned items:{CLOSING_TAG} {files['count']}"

      for file in files['hits']:
        formatted_output += f"\n      {HIGHLIGHT_TAG}↳{CLOSING_TAG} {file}"

      formatted_output += "\n___\n"

    return formatted_output
  
  def format_output_header(self):
    """
    Formats the output report header

    Args:
      None

    Returns:
      A string containing the formatted header
    """
    header = ''
    header += f'\n[Malware Remediation Report]\n'
    header += f'\n------------------------------'
    header += f'\nAuthor:   Website Security Support'
    header += f'\nDate:     {datetime.date.today()}'
    if self.domain:
      header += f"\nDomain:   {self.domain.replace('.', '[.]')}"
    header += f'\n------------------------------\n'
    header += f'\n[Remediation Breakdown]:\n'
    return header
  
  def calculate_percentage(self, family):
    """
    Calculate percentage of which a particular signature family consumes of a total malware infection

    Args:
      family: A string containing a unique signature family

    Returns:
      A float containing the infection percentage
    """
    total_related_sigs = 0

    for sig, file in self.sorted_output:
      if family in sig:
        total_related_sigs += file['count']

    if total_related_sigs == 0 or self.total_artifacts == 0:
      return 0.0

    value = (total_related_sigs / self.total_artifacts) * 100
    percent = round(value, 1) if value >= 10 else round(value, 2)

    return percent
  
  def calculate_percent_bar(self, percent, bar_width=50):
    try:
      filled_length = round((percent / 100) * bar_width)
      empty_length = bar_width - filled_length
    except ValueError as e:
      return '[..........]'
    
    filled_bars = '█' * filled_length
    empty_bars = '.' * empty_length

    return f'[{filled_bars}{empty_bars}]'
  
  def format_output_footer(self, use_highlights):
    """
    Formats the output report footer

    Args:
      use_highlights: Boolean containing the use of highlights

    Returns:
      A string containing the formatted footer
    """
    HIGHLIGHT_TAG, CLOSING_TAG = self._get_tags(use_highlights)

    footer = ''
    footer += '\n[Remediation Statistics]\n'
    # footer += '\n------------------------------'
    footer += f'\n{HIGHLIGHT_TAG}Impacted locations:{CLOSING_TAG} {self.impacted_locations if len(self.impacted_locations) else {}}'
    footer += f'\n{HIGHLIGHT_TAG}Artifacts cleaned:{CLOSING_TAG}  {self.total_artifacts}'
    footer += f'\n{HIGHLIGHT_TAG}Unique signatures:{CLOSING_TAG}  {self.total_signatures}'
    footer += f'\n{HIGHLIGHT_TAG}Signature families:{CLOSING_TAG} {len(self.unique_sig_families)}'

    sorted_sig_family_list = []

    for family in self.unique_sig_families:
      percent = self.calculate_percentage(family)
      sorted_sig_family_list.append((family, percent))

    sorted_sig_family_list.sort(key = lambda n: n[1], reverse=True)

    for family in sorted_sig_family_list:
      sig_family = family[0]
      sig_percent = family[1]
      sig_percent_bar = self.calculate_percent_bar(sig_percent) if self.verbose else ''
      footer += f"\n      {HIGHLIGHT_TAG}↳{CLOSING_TAG} {sig_percent_bar} {sig_percent} % {HIGHLIGHT_TAG}=>{CLOSING_TAG} {sig_family}.*"
    # footer += '\n------------------------------\n'

    
    if self.show_all:
      footer += '\n\n___'
      footer += '\n\n[Miscellaneous]\n'
      misc_output = len(self.other_output_items)
      if misc_output > 0:
        for item in self.other_output_items:
          footer += f"\n{HIGHLIGHT_TAG}-{CLOSING_TAG} {item}"
      else:
        footer += '\nNo additional output'

    footer += '\n\n[End of Report]\n'


    return footer
  
  
  def display_output(self, output):
    """
    Display the formatted output to the user.

    Args:
      output: String containing formatted output.

    Returns:
      None
    """
    print(self.format_output_header())
    print(output)
    print(self.format_output_footer(True))

  def save_output(self, output):
    """
    Save the malware signature details and corresponding file hits to a specified location.

    Args:
      output: String containing formatted output.

    Returns:
      None
    """
    try:
      with open(self.target, 'w') as log:
        log.write(self.format_output_header())
        log.write(output)
        log.write(self.format_output_footer(False))

    except (FileNotFoundError, PermissionError) as e:
      print(f'\n{ERROR_HIGHLIGHT}Could not write parsed output to target file.{CLOSER}\n')
      raise

  def calculate_frequency(self, group, subgroup):
    """
    Generates a matching frequency rating based on group and subgroup values

    Args:
      group:    A string containing the malware signatures group value
      subgroup: A string containing the malware signatures subgroup value

    Returns:
      A string containing the relative frequency value
    """
    if not group: 
      return 'Common'

    try:
      normalizedGroup = int(group)
    except ValueError as e:
      return 'Common'

    if normalizedGroup < RARE_THRESHOLD:
      return f'Rare (# {normalizedGroup})'
    else:
      return f'Common (# {normalizedGroup})'
